#+begin_src java :eval never :tangle ../src/CountOps.java
  // CountOps.java: how often does the innermost statement run?
  public class CountOps
  {
      static long f1(int n) { long c = 0; c++; return c; }
      static long f2(int n) { long c = 0; for (int i = 0; i < n; i++) c++; return c; }
      static long f3(int n) { long c = 0; for (int i = 0; i < n; i++)
                                  for (int j = 0; j < n; j++) c++; return c; }
      static long f4(int n) { long c = 0; for (int i = 0; i < n; i++)
                                  for (int j = 0; j < n; j++)
                                      for (int k = 0; k < n; k++) c++; return c; }
      static long f5(int n) { long c = 0; for (int i = 0; i < n; i++)
                                  for (int j = 0; j < i; j++) c++; return c; }
      static long f6(int n) { long c = 0; for (int i = 1; i < n; i *= 2)
                                  for (int j = 0; j < i; j++) c++; return c; }
      static long f7(int n) { long c = 0; for (int i = n; i > 0; i /= 2) c++; return c; }
      static long f8(int n) { long c = 0; for (int i = 0; i < n; i++)
                                  for (int j = n; j > 0; j /= 2) c++; return c; }

      public static void main(String[] args)
      {
          int n = Integer.parseInt(args[0]);
          System.out.println("n = " + n);
          System.out.printf("%-5s %14s %14s%n", "frag", "n", "2n");
          System.out.printf("%-5d %14d %14d%n", 1, f1(n), f1(2 * n));
          System.out.printf("%-5d %14d %14d%n", 2, f2(n), f2(2 * n));
          System.out.printf("%-5d %14d %14d%n", 3, f3(n), f3(2 * n));
          System.out.printf("%-5d %14d %14d%n", 4, f4(n), f4(2 * n));
          System.out.printf("%-5d %14d %14d%n", 5, f5(n), f5(2 * n));
          System.out.printf("%-5d %14d %14d%n", 6, f6(n), f6(2 * n));
          System.out.printf("%-5d %14d %14d%n", 7, f7(n), f7(2 * n));
          System.out.printf("%-5d %14d %14d%n", 8, f8(n), f8(2 * n));
      }
  }
#+end_src
