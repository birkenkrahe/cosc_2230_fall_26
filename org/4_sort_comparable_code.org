#+begin_src java :tangle InsertionComparable.java
  // InsertionComparable.java: Insertion sort for any Comparable
  // a library (sort) with a test client (main)
  public class InsertionComparable
  {
      // sort a[] into ascending order in place.
      public static void sort(Comparable[] a)
      {
      	// number of entries (input size)
      	int N = a.length;
      	// insert a[i] into the sorted part a[0,i)
      	for (int i = 1; i < N; i++) // first entry already sorted!
      	    // move a[i] to the left, one place per step
      	    for (int j = i; j > 0; j--)
      		// is the left neighbor larger?
      		if (a[j-1].compareTo(a[j]) > 0)
      		    // then swap the two elements
      		    exchange(a, j-1, j);
      	// else a[j] is in place: stop early
      		else break;
      }
      // private helper: only sort() uses it, not in the API
      private static void exchange(Comparable[] a, int i, int j)
      {
      	// swap a[i] and a[j] via the temporary t
      	Comparable t = a[i]; a[i] = a[j]; a[j] = t;
      }
      // InsertionComparable.java: paste this main() inside the class
      // test client: the same sort for strings, integers and doubles
      public static void main(String[] args)
      {
  	// eight words: String implements Comparable
  	String[]  words = { "was", "had", "him", "and", "you", "his", "the", "but" };
  	// six integers: Integer wraps int and implements Comparable
  	Integer[] ints  = { 42, 7, 19, 3, 88, 25 };
  	// six reals: Double wraps double and implements Comparable
  	Double[]  reals = { 2.71, 3.14, 1.41, 0.58 };
  	// one sort() on all three different types
  	sort(words); sort(ints); sort(reals);
  	// print the sorted output as arrays (without hashcode)
  	StdOut.println(java.util.Arrays.toString(words));
  	StdOut.println(java.util.Arrays.toString(ints));
  	StdOut.println(java.util.Arrays.toString(reals));
      }
  }

#+begin_src java :tangle Counter.java :eval-no :no-expand
  // Counter.java - the week 1 Counter now comparable by its count
  public class Counter implements Comparable<Counter>
  {   // Start of `Counter` class
      private final String name;  
      private int count;
      public Counter(String name, int count) { this.name=name; this.count=count; }
      public void increment() { count++; }
      public void decrement() { count--; }
      public int getCount() { return count; }
      public int compareTo(Counter b)
      {
  	// Use Integer.compare on count
  	if (count != b.count) return Integer.compare(count,b.count);
  	// Use compareTo on the name
  	return name.compareTo(b.name);
      }
      // TODO print a Counter as name + count
      public String toString() { return name + " " + count; }
  } // End of `Counter` class
  // test client: read name/votes pairs, sort the counters, print them
  public static void main(String[] args)
  {
      // read all words from standard input: name, votes, name, votes, ...
      String[] in = StdIn.readAllStrings();
      // one counter per name/votes pair
      Counter[] a = new Counter[in.length / 2];
      // for each pair i
      for (int i = 0; i < a.length; i++)
          // a counter named in[2*i], starting at the votes in[2*i+1]
          a[i] = new Counter(in[2*i], Integer.parseInt(in[2*i+1]));
      // sort by votes, ties by name (uses compareTo())
      InsertionGeneric.sort(a);
      // print one counter per line (uses toString())
      for (Counter c : a) StdOut.println(c);
  }
